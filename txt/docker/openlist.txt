docker pull openlistteam/openlist:v4.2.9      #2026-10

------------------------------------------------------

# docker-compose.yml


services:
  openlist:
    image: openlistteam/openlist:v4.2.9
    container_name: openlist
    user: 1000:1000     # Please replace `0:0` with the actual user ID and group ID you want to use to run OpenList.
    volumes:
      - /srv/config/openlist:/opt/openlist/data
    ports:
      - 5244:5244
    environment:
      - UMASK=022
    restart: unless-stopped

# sudo chown -R 1000:1000 /srv/config/alist
------------------------------------------------------

首次运行
docker logs openlist
你将在日志中看到密码。

非首次运行
您可以重新生成或手动设置密码。
# 重新随机生成密码
docker exec -it openlist ./openlist admin random
