cls
@ECHO OFF
CLS
color 0a

GOTO MENU
:MENU
ECHO.
ECHO.               =-=-=-=-=批处理菜单示例=-=-=-=-=
ECHO.
ECHO.                       1  执行命令a
ECHO.
ECHO.                       2  执行命令b
ECHO.
ECHO.                       3  执行命令c
ECHO.
ECHO.                       4  退   出
ECHO.
ECHO.
ECHO.
echo.                请输入选择项目的序号：
set /p  ID=
if "%id%"=="1"  goto cmd1

if "%id%"=="2" goto cmd2

if "%id%"=="3" goto cmd3

IF "%id%"=="4"  exit
PAUSE

:cmd1

echo 执行命令a
rem explorer "http://www.baidu.com"
goto MENU

:cmd2

echo 执行命令b
echo 操作注册表的命令(reg add 主键 值)
GOTO MENU

:cmd3
echo 执行命令c
echo 操作系统服务命令(sc)
GOTO MENU

记事本另存为 ANSI



@echo off
setlocal enabledelayedexpansion
title 主菜单

:MAIN_MENU
cls
echo ============================================
echo                主 菜 单
echo ============================================
echo.
echo   [1] dir
echo   [2] time
echo   [3] ipconfig
echo   [4] 退出
echo.
echo ============================================
set /p "CHOICE=请输入选项 (1-4): "

if "%CHOICE%"=="1" goto OPTION1
if "%CHOICE%"=="2" goto OPTION2
if "%CHOICE%"=="3" goto OPTION3
if "%CHOICE%"=="4" goto EXIT

echo.
echo 输入无效，请重新选择。
timeout /t 2 /nobreak >nul
goto MAIN_MENU

:OPTION1
cls
echo ============================================
echo   [1] 执行 dir
echo ============================================
echo.
dir
echo.
echo 执行完毕，按任意键返回主菜单...
pause >nul
goto MAIN_MENU

:OPTION2
cls
echo ============================================
echo   [2] 执行 time
echo ============================================
echo.
time
echo.
echo 执行完毕，按任意键返回主菜单...
pause >nul
goto MAIN_MENU

:OPTION3
cls
echo ============================================
echo   [3] 执行 ipconfig
echo ============================================
echo.
ipconfig
echo.
echo 执行完毕，按任意键返回主菜单...
pause >nul
goto MAIN_MENU

:EXIT
cls
echo 已退出，再见。
timeout /t 1 /nobreak >nul
exit /b 0
